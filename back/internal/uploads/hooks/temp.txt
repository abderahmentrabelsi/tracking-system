$filePath = "C:/Users/MSI/Desktop/aa.pdf"
$fileName = [System.Convert]::ToBase64String([System.Text.Encoding]::UTF8.GetBytes("aa.pdf"))
$fileSize = (Get-Item $filePath).length
$fileType = [System.Convert]::ToBase64String([System.Text.Encoding]::UTF8.GetBytes("application/pdf"))

# Create the upload resource
$response = curl -i -X POST http://localhost:8383/files/ `
    -H "Tus-Resumable: 1.0.0" `
    -H "Upload-Length: $fileSize" `
    -H "Upload-Metadata: filename $fileName,filetype $fileType"

# Print the response for debugging
Write-Output "Response from server:"
Write-Output $response

# Extract the upload URL
$locationHeader = $response | Select-String -Pattern "Location: (.*)"
if ($locationHeader) {
    $uploadUrl = $locationHeader.Matches.Groups[1].Value.Trim()
    Write-Output "Upload URL: $uploadUrl"
} else {
    Write-Error "Failed to get upload URL from the response."
    exit 1
}

# Upload the file data
curl -X PATCH $uploadUrl `
    -H "Tus-Resumable: 1.0.0" `
    -H "Upload-Offset: 0" `
    -H "Content-Type: application/offset+octet-stream" `
    --data-binary "@$filePath"
